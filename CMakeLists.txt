cmake_minimum_required(VERSION 3.16)
project(wolf C)

set(CMAKE_C_STANDARD 11)

set(MACOS_FRAMEWORK_DIR "" CACHE PATH "Directory containing Universal SDL2 frameworks and headers")

if(APPLE AND MACOS_FRAMEWORK_DIR)
    find_library(SDL2_FRAMEWORK SDL2 PATHS "${MACOS_FRAMEWORK_DIR}" NO_DEFAULT_PATH REQUIRED)
    find_library(SDL2_MIXER_FRAMEWORK SDL2_mixer PATHS "${MACOS_FRAMEWORK_DIR}" NO_DEFAULT_PATH REQUIRED)
else()
    find_package(SDL2 REQUIRED)
    find_package(SDL2_mixer REQUIRED)
endif()

option(DEBUG_NO_AMMO "Start with zero ammo for pistol whip testing" OFF)
option(DEBUG_SHOTGUN "Start with shotgun unlocked for testing" OFF)
option(DEBUG_AK47 "Start with AK47 unlocked for testing" OFF)
set(DEBUG_START_WEAPON "" CACHE STRING "Start with a weapon enum unlocked and equipped (empty = disabled)")
set(DEBUG_START_LEVEL "0" CACHE STRING "Skip to this level on startup (0 = disabled)")

file(GLOB_RECURSE SOURCES src/*.c)
add_executable(wolf ${SOURCES})

if(DEBUG_NO_AMMO)
    target_compile_definitions(wolf PRIVATE DEBUG_NO_AMMO)
endif()
if(DEBUG_SHOTGUN)
    target_compile_definitions(wolf PRIVATE DEBUG_SHOTGUN)
endif()
if(DEBUG_AK47)
    target_compile_definitions(wolf PRIVATE DEBUG_AK47)
endif()
if(NOT DEBUG_START_WEAPON STREQUAL "")
    target_compile_definitions(wolf PRIVATE DEBUG_START_WEAPON=${DEBUG_START_WEAPON})
endif()
if(NOT DEBUG_START_LEVEL STREQUAL "0")
    target_compile_definitions(wolf PRIVATE DEBUG_START_LEVEL=${DEBUG_START_LEVEL})
endif()

if(APPLE AND MACOS_FRAMEWORK_DIR)
    target_include_directories(wolf PRIVATE "${MACOS_FRAMEWORK_DIR}/include" ${CMAKE_SOURCE_DIR}/src)
    target_link_libraries(wolf PRIVATE ${SDL2_FRAMEWORK} ${SDL2_MIXER_FRAMEWORK} m)
else()
    target_include_directories(wolf PRIVATE ${SDL2_INCLUDE_DIRS} ${CMAKE_SOURCE_DIR}/src)
    target_link_libraries(wolf PRIVATE ${SDL2_LIBRARIES} SDL2_mixer::SDL2_mixer m)
endif()
